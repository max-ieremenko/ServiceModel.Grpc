using System.Reflection.Metadata;
using System.Reflection.PortableExecutable;
using NUnit.Framework;

namespace ServiceModel.Grpc.DesignTime.Generator.Test;

[TestFixture]
public class DependenciesTest
{
    [Test]
    public void Test()
    {
        var path = Path.Combine(AppContext.BaseDirectory, "ServiceModel.Grpc.DesignTime.Generators.dll");
        var reader = new PEReader(File.OpenRead(path));
        var test = reader.GetMetadataReader();
        foreach (var item in test.AssemblyReferences)
        {
            var reference = test.GetAssemblyReference(item);
            var name = test.GetString(reference.Name);
            Console.WriteLine($"{name} {reference.Version}");
        }
    }
}
